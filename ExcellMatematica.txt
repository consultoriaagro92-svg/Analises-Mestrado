Funções Básicas

Matemáticas

-- Soma
=SOMA(A1:A10)
=SOMA(A1;A2;A3)

-- Subtração
=A1-A2

-- Multiplicação
=A1*A2

-- Divisão
=A1/A2

-- Potência
= POTENCIA(2;3)
=2^3

-- Raiz Quadrada
=RAIZ(16)

-- Arredondamento
=ARREDONDA(3,14159;2)
=ARREDONDA(3,14159;0)

-- Módulo (resto da divisão)

-- Absoluto
=ABS(-10)

-- Máximo e Mínimo
=MAX(A1:A10)
=MIN(A1:A10)

Lógicas

-- SE
=SE(A1>10;"Alto";"Baixo")

-- OU (OR)

Exemplos Práticos
1. Sistema de Notas

-- Nota final
=(A1*0,4)+(B1*0,3)+(C1*0,3)

-- Status do aluno
=SE(A1>=7;"Aprovado";SE(A1>=5;"Recuperação";"Reprovado"))

-- Média da turma
=MÉDIA(A1:A50)

-- Aprovados
=CONTAR.SE(A1:A50;">=7")

2. Sistema de Vendas

-- Preço com desconto
=B1*(1-C1)

-- Total da venda
=SOMA(D1:D10)

-- Vendas por vendedor
=SOMASE(Vendedores;"João";Vendas)

-- Melhor vendedor
=MÁXIMO.SE(Vendedores;"João";Vendas)

Sistema Financeiro

-- Juros compostos
=FV(0,1;1000;100;-1000)

-- Valor presente
=PV(0,1;1000;100;-1000)

-- Taxa de retorno
=TIR(A1:A10)

-- Custo médio
=MÉDIA.SE(Custos;"<100")

4. Sistema de Estoque

-- Estoque atual
=SOMA(Entradas)-SOMA(Saídas)

-- Valor do estoque
=SOMA(PRODUTO(Estoq;Preços))

-- Estoque baixo
=CONTAR.SE(Estoq;"<10")

-- Preço médio
=MÉDIA.SE(Preços;"<100")

5. Sistema de RH

-- Idade
=DIFERENÇA.ANOS(A1;HOJE())

-- Tempo de empresa
=DIFERENÇA.ANOS(A1;HOJE())

-- Salário médio por departamento
=MÉDIA.SE(Departamento;"Vendas";Salários)

-- Funcionários por departamento
=CONTAR.SE(Departamento;"Vendas")
