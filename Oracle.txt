CREATE TABLE usuarios (
    id NUMBER PRIMARY KEY,
    nome VARCHAR2(100) NOT NULL,
    email VARCHAR2(100) UNIQUE NOT NULL,
    idade NUMBER CHECK (idade >= 0),
    data_nascimento DATE,
    ativo CHAR(1) DEFAULT 'S' CHECK (ativo IN ('S', 'N')),

    CONSTRAINT chk_idade CHECK (idade >= 18),
    CONSTRAINT fk_departamento FOREIGN KEY (departamento_id) REFERENCES departamentos(id)

Sistema de Vendas


-- Criar tabelas
CREATE TABLE produtos (
    id NUMBER PRIMARY KEY,
    nome VARCHAR2(100) NOT NULL,
    prec NUMBER(10,2) NOT NULL
);

CREATE TABLE vendas (
    id NUMBER PRIMARY KEY,
    produto_id NUMBER,
    quantidade NUMBER,
    data_venda DATE DEFAULT SYSDATE,
    FOREIGN KEY (produto_id) REFERENCES produtos(id)
);

-- Consulta de vendas por produto
SELECT p.nome as produto,
       COUNT(v.id) as total_vendas,
       SUM(v.quantidade) as total_quantidade,
       SUM(v.quantidade * p.prec) as valor_total
FROM produtos p
LEFT JOIN vendas v ON p.id = v.produto_id
GROUP BY p.id, p.nome
ORDER BY valor_total DESC;

Sistema Financeiro


-- Tabela de transações
CREATE TABLE transacoes (
    id NUMBER PRIMARY KEY,
    tipo VARCHAR2(20) CHECK (tipo IN ('ENTRADA', 'SAIDA')),
    valor NUMBER(10,2) NOT NULL,
    categoria VARCHAR2(50),
    data_transacao DATE DEFAULT SYSDATE
);

-- Saldo atual
SELECT 
    SUM(CASE WHEN tipo = 'ENTRADA' THEN valor ELSE 0 END) as total_entradas,
    SUM(CASE WHEN tipo = 'SAIDA' THEN valor ELSE 0 END) as total_saidas,
    SUM(CASE WHEN tipo = 'ENTRADA' THEN valor ELSE -valor END) as saldo
FROM transacoes;

-- Transações por mês
SELECT 
    TO_CHAR(data_transacao, 'YYYY-MM') as mes,
    COUNT(*) as total_transacoes,
    SUM(CASE WHEN tipo = 'ENTRADA' THEN valor ELSE 0 END) as entradas,
    SUM(CASE WHEN tipo = 'SAIDA' THEN valor ELSE 0 END) as saidas
FROM transacoes
GROUP BY TO_CHAR(data_transacao, 'YYYY-MM')
ORDER BY mes;

Sistema de Estoque


-- Tabela de estoque
CREATE TABLE estoque (
    produto_id NUMBER PRIMARY KEY,
    quantidade NUMBER DEFAULT 0,
    ultima-atualização TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (produto_id) REFERENCES produtos(id)
);

-- Movimentações
CREATE TABLE movimentacoes (
    id NUMBER PRIMARY KEY,
    produto_id NUMBER,
    tipo VARCHAR2(20) CHECK (tipo IN ('ENTRADA', 'SAIDA')),
    quantidade NUMBER NOT NULL,
    data_movimento TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (produto_id) REFERENCES produtos(id)
);

-- Atualizar estoque
UPDATE estoque 
SET quantidade = quantidade + (
    SELECT CASE WHEN tipo = 'ENTRADA' THEN quantidade ELSE -quantidade END
    FROM movimentacoes 
    WHERE produto_id = :produto_id AND id = :movimento_id
)
WHERE produto_id = :produto_id;

-- Relatório de estoque
SELECT p.nome as produto,
       e.quantidade as estoque atual,
       COALESCE(SUM(CASE WHEN m.tipo = 'ENTRADA' THEN m.quantidade ELSE 0 END), 0) as entradas,
       COALESCE(SUM(CASE WHEN m.tipo = 'SAIDA' THEN m.quantidade ELSE 0 END), 0) as saidas
FROM produtos p
LEFT JOIN estoque e ON p.id = e.produto_id
LEFT JOIN movimentacoes m ON p.id = m.produto_id
GROUP BY p.id, p.nome, e.quantidade;

Sistema de RH


-- Tabela de funcionarios
CREATE TABLE funcionarios (
    id NUMBER PRIMARY KEY,
    nome VARCHAR2(100) NOT NULL,
    cargo VARCHAR2(50),
    departamento_id NUMBER,
    data_contratacao DATE DEFAULT SYSDATE,
    salário NUMBER(10,2),
    FOREIGN KEY (departamento_id) REFERENCES departamentos(id)
);

-- Consulta de funcionarios por departamento
SELECT d.nome as departamento,
       COUNT(f.id) as total_funcionarios,
       AVG(f.salário) as salário_medio,
       MIN(f.salário) as salário_minimo,
       MAX(f.salário) as salário_maximo
FROM departamentos d
LEFT JOIN funcionarios f ON d.id = f.departamento_id
GROUP BY d.id, d.nome
ORDER BY total_funcionarios DESC;

-- Anos de empresa
SELECT nome, 
       EXTRACT(YEAR FROM SYSDATE) - EXTRACT(YEAR FROM data_contratacao) as anos empresa
FROM funcionarios;

Funções de Série Temporal


-- Análise de vendas por período
SELECT 
    TO_CHAR(data_venda, 'YYYY-MM') as periodo,
    COUNT(*) as total_vendas,
    SUM(quantidade * prec) as valor_total,
    AVG(quantidade * prec) as ticket_medio
FROM vendas v
JOIN produtos p ON v.produto_id = p.id
GROUP BY TO_CHAR(data_venda, 'YYYY-MM')
ORDER BY periodo;

-- Comparação com período anterior
WITH vendas_periodo AS (
    SELECT 
        TO_CHAR(data_venda, 'YYYY-MM') as periodo,
        SUM(quantidade * prec) as total
    FROM vendas v
    JOIN produtos p ON v.produto_id = p.id
    GROUP BY TO_CHAR(data_venda, 'YYYY-MM')
)
SELECT 
    vp.periodo,
    vp.total as periodo atual,
    LAG(vp.total) OVER (ORDER BY vp.periodo) as periodo_anterior,
    ROUND((vp.total - LAG(vp.total) OVER (ORDER BY vp.periodo)) / 
          LAG(vp.total) OVER (ORDER BY vp.periodo) * 100, 2) as variacao_percentual
FROM vendas_periodo vp
ORDER BY vp.periodo;

Funções de Análise Espacial (Oracle Spatial)


-- Criar tabela com dados espaciais
CREATE TABLE lojas (
    id NUMBER PRIMARY KEY,
    nome VARCHAR2(100),
    localização SDO_GEOMETRY
);

-- Inserir dados espaciais
INSERT INTO lojas (id, nome, localização) VALUES (
    1, 
    'Loja Centro', 
    SDO_GEOMETRY(
        2001,  -- Tipos de geometria: 2001=ponto, 2002=linha, 2003=polígono
        8377,  -- SRID (Sistema de Referência de Coordenadas)
        SDO_POINT_TYPE(-23.5, -46.6, NULL)  -- Coordenadas
    )
);

-- Consulta espacial
SELECT l1.nome as loja1, l2.nome as loja2,
       SDO_GEOM.SDO_DISTANCE(l1.localização, l2.localização, 0.5) as distancia
FROM lojas l1, lojas l2
WHERE l1.id != l2.id
  AND SDO_GEOM.SDO_DISTANCE(l1.localização, l2.localização, 0.5) < 1000;  -- 1km

Exemplos de Relatórios Complexos

Relatório de Vendas com Análise de Tendência


WITH vendas_diarias AS (
    SELECT 
        TRUNC(data_venda) as data,
        SUM(quantidade * prec) as vendas
    FROM vendas v
    JOIN produtos p ON v.produto_id = p.id
    GROUP BY TRUNC(data_venda)
),
media_movel AS (
    SELECT 
        data,
        vendas,
        AVG(vendas) OVER (ORDER BY data ROWS BETWEEN 6 PRECEDING AND CURRENT ROW) as media_movel_7dias
    FROM vendas_diarias
)
SELECT 
    data,
    vendas,
    media_movel_7dias,
    ROUND((vendas - media_movel_7dias) / media_movel_7dias * 100, 2) as variacao_percentual
FROM media_movel
ORDER BY data;

Análise de Customer Lifetime Value (CLV)

WITH vendas_por_Cliente AS (
    SELECT 
        cliente_id,
        COUNT(*) as total_compras,
        SUM(quantidade * prec) as valor_total,
        AVG(quantidade * prec) as valor_medio,
        MIN(data_venda) as primeira_compra,
        MAX(data_venda) as ultima_compra
    FROM vendas v
    JOIN produtos p ON v.produto_id = p.id
    GROUP BY cliente_id
)
SELECT 
    cliente_id,
    total_compras,
    valor_total,
    valor_medio,
    MONTHS_BETWEEN(ultima_compra, primeira_compra) + 1 as meses_comoCliente,
    ROUND((valor_total / (MONTHS_BETWEEN(ultima_compra, primeira_compra) + 1)), 2) as clv_mensal
FROM vendas_por_Cliente
ORDER BY valor_total DESC;


Dashboard Executivo


-- KPIs principais
SELECT 
    (SELECT COUNT(*) FROM usuarios WHERE ativo = 'S') as usuarios_ativos,
    (SELECT SUM(quantidade * prec) FROM vendas WHERE data_venda >= TRUNC(SYSDATE, 'MM')) as vendas_mes,
    (SELECT COUNT(*) FROM pedidos WHERE status = 'PENDENTE') as pedidos_pendentes,
    (SELECT AVG(quantidade * prec) FROM vendas WHERE data_venda >= TRUNC(SYSDATE, 'MM')) as ticket_medio;

-- Vendas por produto (top 5)
SELECT *
FROM (
    SELECT 
        p.nome as produto,
        SUM(v.quantidade * p.prec) as valor_total,
        ROW_NUMBER() OVER (ORDER BY SUM(v.quantidade * p.prec) DESC) as rank
    FROM produtos p
    JOIN vendas v ON p.id = v.produto_id
    WHERE v.data_venda >= ADD_MONTHS(SYSDATE, -3)
    GROUP BY p.id, p.nome
)
WHERE rank <= 5;
