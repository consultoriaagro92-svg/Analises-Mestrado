Tratamento de Exceções

try/catch


<?php
function dividir($a, $b) {
    if ($b == 0) {
        throw new Exception("Divisão por zero não é permitida");
    }
    return $a / $b;
}

try {
    $resultado = dividir(10, 0);
    echo "Resultado: $resultado";
} catch (Exception $e) {
    echo "Erro: " . $e->getMessage();
} finally {
    echo "Isso sempre será executado";
}

// Exceções customizadas
class CustomException extends Exception {
    public function __construct($message, $code = 0, Exception $previous = null) {
        parent::__construct($message, $code, $previous);
    }
    
    public function __toString() {
        return __CLASS__ . ": {$this->getMessage()}\n";
    }
}

try {
    throw new CustomException("Erro customizado");
} catch (CustomException $e) {
    echo $e;
}
?>

PHP com Validação de Dados



<?php
function validateEmail($email) {
    return filter_var($email, FILTER_VALIDATE_EMAIL) !== false;
}

function validate CPF($cpf) {
    // Remover formatação
    $cpf = preg_replace('/[^0-9]/', '', $cpf);
    
    // Verificar se tem 11 dígitos
    if (strlen($cpf) != 11) {
        return false;
    }
    
    // Verificar se não é sequência de números iguais
    if (preg_match('/^(\d)\1{10}$/', $cpf)) {
        return false;
    }
    
    // Validar dígitos verificadores
    for ($t = 9; $t <= 10; $t++) {
        for ($c = 0, $s = 0; $c < $t; $c++) {
            $s += $cpf[$c] * (($t + 1) - $c);
        }
        if ($cpf[$t] == ((($s * 10) % 11) % 10)) {
            continue;
        }
        return false;
    }
    
    return true;
}

// Usar as validações
if (validateEmail('joao@email.com')) {
    echo "Email válido";
}

if (validate CPF('123.456.789-09')) {
    echo "CPF válido";
}
?>

API RESTful


<?php
header('Content-Type: application/json');

$method = $_SERVER['REQUEST_METHOD'];
$pdo = new PDO("mysql:host=localhost;dbname=app", 'user', 'pass');

switch ($method) {
    case 'GET':
        $id = $_GET['id'] ?? null;
        if ($id) {
            $stmt = $pdo->prepare("SELECT * FROM items WHERE id = ?");
            $stmt->execute([$id]);
            $item = $stmt->fetch();
            echo json_encode($item);
        } else {
            $stmt = $pdo->query("SELECT * FROM items");
            $items = $stmt->fetchAll();
            echo json_encode($items);
        }
        break;
        
    case 'POST':
        $data = json_decode(file_get_contents("php://input"));
        $stmt = $pdo->prepare("INSERT INTO items (nome) VALUES (?)");
        $stmt->execute([$data->nome]);
        echo json_encode(['success' => true]);
        break;
        
    case 'PUT':
        $data = json_decode(file_get_contents("php://input"));
        $stmt = $pdo->prepare("UPDATE items SET nome = ? WHERE id = ?");
        $stmt->execute([$data->nome, $data->id]);
        echo json_encode(['success' => true]);
        break;
        
    case 'DELETE':
        $id = $_GET['id'];
        $stmt = $pdo->prepare("DELETE FROM items WHERE id = ?");
        $stmt->execute([$id]);
        echo json_encode(['success' => true]);
        break;
}
?>
